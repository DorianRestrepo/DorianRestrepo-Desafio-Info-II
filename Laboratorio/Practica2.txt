// Para el case 3 
int largo(const char* cadena) {
    int longitud = 0;
    while (cadena[longitud] != '\0') {
        longitud++;
    }
    return longitud;
}

bool igualdad(const char* cad1, const char* cad2) {
    int len1 = largo(cad1);
    int len2 = largo(cad2);

    if (len1 != len2) {
        return false;
    }

    for (int i = 0; i < len1; i++) {
        if (cad1[i] != cad2[i]) {
            return false;
        }
    }

    return true; 
}

void convertir(int numero, char* cadena) {
    int i = 0;

    // Caso especial para el cero
    if (numero == 0) {
        cadena[0] = '0';  
        cadena[1] = '\0'; 
        return;
    }

    bool negativo = false;
    
    if (numero < 0) {
        negativo = true;
        numero = -numero;
    }

   
    while (numero > 0) {
        cadena[i++] = (numero % 10) + '0'; // Conversion a codigo ASCII
        numero /= 10;
    }

    if (negativo) {
        cadena[i++] = '-';
    }

    cadena[i] = '\0'; 

    // Invierte el arreglo sobre su propio espacio de memoria
    int inicio = 0;
    int fin = i - 1;
    while (inicio < fin) {
        char temp = cadena[inicio];
        cadena[inicio] = cadena[fin];
        cadena[fin] = temp;
        inicio++;
        fin--;
    }
}
